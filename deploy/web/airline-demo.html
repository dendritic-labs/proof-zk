🔧 Integration Preview

See how easy it is to integrate ProofZK into your existing website. Just a few lines of code! 

<!-- Add to your airline website -->
<script src="https://demo.proofzk.com/sdk/proofzk.js"></script>

<button onclick="proofzkCheckin()">ProofZK Quick Check-in</button>

<script>
async function proofzkCheckin() {
    // Request age verification (18+)
    const proof = await ProofZK.requestProof({
        type: 'age_verification',
        minimum_age: 18,
        purpose: 'airline_checkin'
    });
    
    if (proof.verified) {
        // Customer is 18+ (no other data revealed)
        completeCheckin();
    }
}
</script>
